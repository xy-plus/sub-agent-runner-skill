# 只给 writer 一个身份 —— 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让「这一轮真的结束了没有」问对进程。续跑之前要确认的是**没有人还会往这份日志里写字**，而真正在写的是**包装器**（tee 循环读 codex 的 stdout 管道，codex 的后代继承了同一个管道，codex 死了管道不 EOF）。今天 `wait_for_exit` 等的却是 codex，而 `find_codex_pid` 用 argv 精确比对、僵尸的 `cmdline` 为空，于是 codex **变僵尸那一刻**它就返回 True——实测那时日志还要再长 3 秒：

```
t=0.000s  codex=僵尸（cmdline 已空 → find_codex_pid 判 None）  writer=活  ← 此刻已返回 True，日志还在长
t=3.007s  writer 才退出
```

**Architecture:** **只把 writer 的身份记下来，codex 那一侧一个字不改。** `writer_pid` = `os.getpid()`，Popen 之前就知道 → **一次 `write_meta` 写完**，不存在「写到一半 writer 死了」的失败面。身份是 **PID + `/proc/<pid>/stat` 第 22 字段（启动时刻）**，都是 `str`。等待条件从「codex 没了」改成**两个都停了**：`writer_still_running`（还有人写日志吗）与 `find_codex_pid`（会话还被占着吗，孤儿 codex 走这条）。writer 那半用 `os.pidfd_open` + `select` 阻塞等，codex 那半继续轮询扫 `/proc`——它刻意没有记下来的身份，现场事实不依赖「有人记得写」。

**Tech Stack:** Python 3 stdlib。`codex_agent.py` **加 `import select`**（`import time` 保留：`wait_until_quiet` 的 deadline 和 codex 那半的轮询都还用它）；`test_codex_agent.py` **加 `import ctypes`**（`prctl(PR_SET_NAME)` 造畸形 `comm`）。零第三方依赖。

```python
class TestWriterIdentity(unittest.TestCase):
    """writer 的身份 = PID + `/proc/<pid>/stat` 第 22 字段（启动时刻）。

    为什么只有 writer 记身份：codex 的存活有**现场特征**可查（comm=codex、argv
    里有报告路径），而「还有没有人往日志里写字」一点现场特征都没有——日志停不停
    长看不出来，管道 EOF 归 stdout 继承关系管。所以链上唯一会死的那个进程必须
    自报身份，而它在 Popen 之前就知道自己是谁，一次写完。

    裸 PID 判不了两件事，而这两件事正是这一轮 bug 的全部：
      ① 僵尸——`os.kill(pid, 0)` 对僵尸**成功返回**，说「活着」
      ② PID 复用——同一个 PID 换了个进程，裸 PID 一样说「活着」
    """

    @staticmethod
    def _meta_of(pid, start):
        return {"writer_pid": pid, "writer_start": start}

    def test_活着的writer算还在写(self):
        pid, start = ca.writer_identity()
        self.assertTrue(ca.writer_still_running(self._meta_of(pid, start)))

    def test_身份成对返回_两半都是str(self):
        # 两半必须同时拿到：分开取就会出现「记了 PID 没记启动时刻」的半截身份，
        # 而半截身份等于退回裸 PID。
        ident = ca.writer_identity()
        self.assertEqual(len(ident), 2)
        self.assertEqual(ident[0], str(os.getpid()))
        self.assertTrue(all(isinstance(v, str) for v in ident),
                        f"身份必须是 str：{ident!r}——比对是字符串比对，int 会把活着的任务判死")

    def test_启动时刻取的是stat的第22字段(self):
        """判据是**独立算出来的**，不是拿被测函数自己的切法当参照。

        第 22 字段是「开机以来的滴答数」，所以「进程年龄 = 开机至今 − 启动时刻」
        必须落在一个很小的正数区间里。左右两个邻居都被这条挡住：
        第 21 字段 itrealvalue 恒为 0（年龄 = 整个开机时长），
        第 23 字段 vsize 是字节数（2026-09-20 实测 18,898,944，年龄算出来是负的几万秒）。
        """
        滴答每秒 = os.sysconf("SC_CLK_TCK")
        开机至今 = int(float(pathlib.Path("/proc/uptime").read_text().split()[0]) * 滴答每秒)
        启动时刻 = int(ca.writer_identity()[1])
        self.assertGreater(启动时刻, 0, "第 21 字段(itrealvalue)恒为 0——下标写小了")
        年龄 = 开机至今 - 启动时刻
        self.assertGreaterEqual(年龄, 0, "启动时刻比开机至今还晚，切到别的字段上去了")
        self.assertLess(年龄, 300 * 滴答每秒,
                        "本测试进程不可能活了 5 分钟以上（整套测试实跑 ~2 秒）——下标写大了")

    def test_comm含空格括号制表符和裸换行时仍然解析正确(self):
        """`comm` 是进程自己用 `prctl(PR_SET_NAME)` 设的**任意 15 字节**。

        2026-09-20 实测 comm = `we ird)\\nx` 时 `/proc/<pid>/stat` **按行读会得到 2 行**，
        而第一行里最后一个 `)` 落在 comm **内部**——按行读切出来的字段列表长度是 **0**，
        下标 19 当场 IndexError；就算侥幸不炸，切出来的「启动时刻」也是 comm 的碎片。
        这种瞎法很安静：只在别人给进程改过名的时候才发作。
        参照物是**改名之前**读到的启动时刻，它由独立的一次读取得到。
        """
        陪练 = subprocess.Popen(
            [sys.executable, "-c",
             "import ctypes, sys, time\n"
             "sys.stdout.write('before\\n'); sys.stdout.flush()\n"
             "sys.stdin.readline()\n"
             "ctypes.CDLL('libc.so.6').prctl(15, b'we ird)\\nx', 0, 0, 0)\n"
             "sys.stdout.write('after\\n'); sys.stdout.flush()\n"
             "time.sleep(30)\n"],
            stdin=subprocess.PIPE, stdout=subprocess.PIPE)
        try:
            self.assertEqual(陪练.stdout.readline(), b"before\n", "前提不成立：陪练没起来")
            改名前 = ca._read_stat_fields(str(陪练.pid))
            self.assertIsNotNone(改名前, "前提不成立：陪练的 /proc 读不到")
            陪练.stdin.write(b"go\n")
            陪练.stdin.flush()
            self.assertEqual(陪练.stdout.readline(), b"after\n", "前提不成立：陪练没改成名")
            raw = pathlib.Path(f"/proc/{陪练.pid}/stat").read_bytes()
            self.assertIn(b"\n", raw[:raw.rindex(b")")],
                          "前提不成立：comm 里没有裸换行，这条测不到「不许按行读」")
            self.assertEqual(len(raw.splitlines()), 2,
                             "前提不成立：这个文件按行读只有一行，那按行读的突变杀不掉")
            self.assertEqual(ca._read_stat_fields(str(陪练.pid)), 改名前,
                             "comm 里的空格/括号/制表符/裸换行把解析带偏了")
        finally:
            陪练.kill()
            陪练.wait()
            陪练.stdin.close()
            陪练.stdout.close()

    def test_进程不在了就算停了(self):
        self.assertIsNone(ca._read_stat_fields(GONE_PID))
        self.assertFalse(ca.writer_still_running(self._meta_of(GONE_PID, GONE_START)))

    def test_同一个PID但启动时刻对不上就算停了_这是防PID复用的那一道(self):
        # 自然情况下撞同一副 (PID, 10ms 刻度) 的余量是 445,255 倍，造不出来也不该造。
        # 但**比对逻辑本身**两行就测完了，而它正是那道防线。
        pid, start = ca.writer_identity()
        self.assertTrue(ca.writer_still_running(self._meta_of(pid, start)),
                        "前提不成立：自己居然不算在写")
        self.assertFalse(ca.writer_still_running(self._meta_of(pid, str(int(start) + 1))))

    def test_writer变僵尸就算停了_僵尸不可能再写字(self):
        """僵尸已经不执行任何代码，日志不可能再长。

        **不许等它被回收**：回收时机归 harness 的 bash/node 管，那是第三方——
        spec 的版本小史里 v4 就栽在这一条上。
        这条测试里**不许再建第二个 `subprocess.Popen`**：`Popen.__init__` 会调
        `subprocess._cleanup()`，顺手把这个僵尸收掉，测试于是测到另一条分支上去。
        """
        僵尸 = subprocess.Popen([sys.executable, "-c", "pass"])
        try:
            start = None
            deadline = time.time() + 5
            while time.time() < deadline:
                fields = ca._read_stat_fields(str(僵尸.pid))
                if fields is not None and fields[0] == "Z":
                    start = fields[1]
                    break
                time.sleep(0.01)
            self.assertIsNotNone(start, f"陪练 {僵尸.pid} 始终没变成僵尸，本测试验证不了任何东西")
            os.kill(僵尸.pid, 0)   # 前提：裸 PID 对僵尸「成功」，它判不出来；抛异常就是前提变了
            self.assertFalse(ca.writer_still_running(self._meta_of(str(僵尸.pid), start)))
        finally:
            僵尸.wait()
```

```python
# **PID 复用的余量是 445,255 倍，所以这里零行代码去防它。** 启动时刻的分辨率是
# 10ms（CLK_TCK=100），要让两副身份撞车就得在同一个 10ms 刻度内把同一个 PID 发两次，
# 而 PID 是顺序分配、绕完 pid_max 才回头：
#     pid_max 4,194,304 → 10ms 内绕完一圈需 4.19 亿次 fork/秒
#     本机实测 fork 速率 942 次/秒（绕一圈 1.2 小时）→ 余量 445,255 倍
# **唯一真实的边界**：别拿这套身份去判一个活不到 10ms 的短命进程。
# 本工具记身份的只有 writer，它陪跑整轮，分钟到小时量级。
def _read_stat_fields(pid):
    """`/proc/<pid>/stat` 里的 `(运行状态, 启动时刻)`，都是 `str`；进程不在了返回 `None`。

    **整文件读 bytes，从最后一个 `)` 之后切。禁止按行读、禁止 split() 全文。**
    `comm` 是进程自己用 `prctl(PR_SET_NAME)` 设的**任意 15 字节**：空格、括号、
    制表符、**裸换行**全都进得去（2026-09-20 实测 comm=`we ird)\\nx`，那时这个文件
    **按行读得到 2 行**，而第一行里最后一个 `)` 落在 comm 内部，切出来的字段列表
    长度是 0，下标 19 当场 IndexError）。它瞎得很安静——只在别人给进程改过名时才发作。

    `tail[0]` 是 stat 的第 3 字段（state），`tail[19]` 是第 22 字段（starttime）：
    前两个字段 pid 和 comm 已经被切掉了，所以下标是 `22 - 2 - 1`。

    `pid` 收 `str`：元数据里存的就是 `str`，全模块只留这一种传法。
    """
    try:
        raw = pathlib.Path(f"/proc/{pid}/stat").read_bytes()
    except OSError:
        # 进程随时可能退出（ENOENT），/proc 也可能读到一半没了——一律当「不在了」
        return None
    tail = raw[raw.rindex(b")") + 1:].split()
    return tail[0].decode(), tail[19].decode()


def writer_identity():
    """**本进程**的身份 `(pid, 启动时刻)`，落进元数据给下一条命令比对。

    **成对返回，拿不到半截**：半截身份等于退回裸 PID，而裸 PID 判不出僵尸也挡不住
    复用——那正是这一轮要修的。

    只有 writer 记身份，codex 那侧继续用 `find_codex_pid` 现场反查。分界线是
    **有没有现场特征**：codex 有（comm=codex、argv 里有报告路径），而「还有没有人
    往日志里写字」没有任何现场特征。现场事实不依赖「有人记得写下来」，而 writer
    正是这条链上唯一会死的那个进程——所以只有它必须自报。

    读不到自己就直接炸，不返回 `None` 让调用方去猜：`/proc` 读不到自己意味着本工具
    的全部存活判定（`find_codex_pid` 也在内）都不成立，没有第二条路可走。
    """
    pid = str(os.getpid())
    fields = _read_stat_fields(pid)
    if fields is None:
        raise RuntimeError(f"/proc/{pid}/stat 读不到自己——本工具的全部存活判定都架在 /proc 上")
    return pid, fields[1]


def writer_still_running(meta):
    """还有人在往这份日志里写字吗——问的是**包装器**，不是 codex。

    身份 = PID + 启动时刻，缺一不可。实测（PID namespace 强制复用）：裸
    `os.kill(pid, 0)` 说「活着」，比对启动时刻正确判「已退出」。

    **僵尸算停了**：它已经不执行任何代码，日志不可能再长；等它被父进程回收要靠
    harness 的 bash/node，那是第三方，不能当判据。

    `find_codex_pid` 的 uid 过滤是承重的（本机有别的用户在跑 codex，不限用户会
    打到别人身上）。writer 这一侧**顶上来的是启动时刻比对，不是 uid**——pidfd
    不做权限过滤（实测能 open root 的 PID 1），所以这条替换关系要写下来，
    否则下一个人会以为那边的 uid 过滤是白留的。
    """
    fields = _read_stat_fields(meta["writer_pid"])
    if fields is None:
        return False
    state, start = fields
    return start == meta["writer_start"] and state != "Z"
```

```python
# 打断之后最多等这一轮安静下来多久。**这是策略值，不是实测上界。**
# 排干多久由 codex 起的后代决定，**没有上界**——它起个后台服务就永不结束。
# 所以这个数字回答的是「本工具愿意等多久」：够大到不会误杀正常收尾
# （2026-09-19 实测 INT → codex 消失是 1.854 秒和 0.964 秒，writer 再多几秒），
# 够小到卡住时调用方不会被无限期挂着。**不要把它当成测量值去钉倍数余量。**
#
# 必须是**模块级常量**而不是函数默认参数：测试要把它压到 0.3 秒；
# 也正合仓库规范「不要默认缺省值」——调用方每次都显式传，不可能漏。
INTERRUPT_EXIT_TIMEOUT = 60
INTERRUPT_POLL_INTERVAL = 0.2


def _wait_writer_stops(meta, deadline):
    """阻塞等包装器停笔，返回是否在 `deadline`（`time.monotonic()` 刻度）之前停下。

    用 `os.pidfd_open` + `select`：pidfd 在**终止**那一刻就可读、**不等父进程
    回收**（实测 Linux 6.8.0 / Python 3.12：非子进程上照样精确唤醒，唤醒时
    `/proc/<pid>/stat` 的状态还是 `Z`）。等「被回收」要靠 harness 的 bash/node，
    那是第三方，不能当判据。

    **可用性前提：内核 ≥5.3。** `os.pidfd_open` 这个属性**存在是编译期性质**、
    `ENOSYS` 是**运行期**性质，`hasattr` 挡不住——所以这里不写 hasattr，
    让失败落到下面那句读得懂的拒绝上。
    **刻意不做轮询退化**：两份存活判据必然漂移，而漂移的后果是静默的。
    """
    # 元数据里存的是 str（json 往返不保类型，见 _load_meta）；
    # 实测 os.kill("123", 0) / os.pidfd_open("123") 都是直接 TypeError，
    # 所以还原成 int 的落地点只有这一处。
    pid = int(meta["writer_pid"])
    try:
        fd = os.pidfd_open(pid)
    except ProcessLookupError:
        return True                    # 连 fd 都开不出来，它早没了
    except OSError as e:
        reject(f"os.pidfd_open({pid}) 失败：{e}。本工具靠 pidfd 等包装器停笔，"
               f"它要内核 ≥5.3。这一条没有退化路径：再加一份轮询判据，"
               f"两份必然漂移，而漂移是静默的。")
    try:
        # **复校必须排在 select 之前。** open 到这里之间 PID 可能已经被复用，
        # 写反的代价实测过：在一个**无关的复用者**身上卡满整个超时
        # （探针里 3.01s，真实代码是 60s）。复校不匹配就直接走，绝不等。
        #
        # TOCTOU 由此闭合，而闭合的理由**不是**「fd 锁住了 PID 号」——实测持有
        # pidfd **并不**锁住 PID：老 fd 仍正确指向老进程，而 /proc/<pid> 已经是
        # 新进程的了。真正的理由是：open 之后才可能发生的复用，**蕴含**原进程
        # 已被回收，而那时「它不在写了」本来就是正确答案。
        if not writer_still_running(meta):
            return True
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            return False
        return bool(select.select([fd], [], [], remaining)[0])
    finally:
        os.close(fd)


def wait_until_quiet(meta, report_path, timeout, poll_interval):
    """等到**没人再往这份日志里写字**、而且**没人再占着这个会话**。返回是否等到了。

    两个条件问的是两件事，缺一不可：

        还有人写日志吗   writer_still_running   包装器，身份记在元数据里
        会话还被占着吗   find_codex_pid         现场反查

    只看 codex 会在它**变僵尸那一刻**就返回（僵尸的 cmdline 为空，argv 比对拒绝），
    而实测那时日志还要再长 3 秒（t=0.000s codex 僵尸 → t=3.007s writer 才退出）。
    只看包装器会撞上**孤儿 codex**：SIGKILL 掉包装器之后 codex 存活并跑完（实测），
    续跑就撞上它的写锁——而那正是 `interrupt-and-resume` 存在的理由。

    `report_path` 是 `pathlib.Path`，和本模块其它地方一致（见 find_codex_pid）。
    **它由调用方传进来，不从 meta 反推**：目录布局归 `isolation_home`／`_report_path`
    管，等待函数不该知道产物落在哪儿。

    这是 `interrupt-and-resume` **唯一独有的收益**：护栏只会拒绝，不会替你等。
    `cmd_resume` 早就拦住了「对还在跑的会话 resume」（实测 stop 之后 0.164 秒
    resume，拿到的是干净的 exit 2 拒绝，不是 thread-store conflict），但调用方
    拿到 exit 2 之后得自己写重试循环——间隔多少、上界多少、超时了怎么办，全是
    软约定，每个调用方现编一遍，编错了没人告诉他。

    **只等，不发任何信号。** 超时的正确处置是让调用方稍后再来，不是加大火力：
    升级到 SIGTERM 会让会话永久锁死，而那一步不可逆。
    """
    deadline = time.monotonic() + timeout
    if not _wait_writer_stops(meta, deadline):
        return False
    # codex 这一侧**刻意没有**记下来的身份：它有现场特征可查，而现场事实不依赖
    # 「有人记得写下来」（见 writer_identity）。代价是只能轮询扫 /proc——
    # 整体唤醒粒度因此本来就是 poll_interval 这一档，writer 那半等得再精确
    # 也快不过它。
    while True:
        if find_codex_pid(report_path) is None:
            return True
        if time.monotonic() >= deadline:
            return False
        time.sleep(poll_interval)
```
